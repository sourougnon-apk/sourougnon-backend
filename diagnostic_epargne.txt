--- app/Services/EpargneService.php ---
<?php
namespace App\Services;
use App\Models\Vente;
use App\Models\Recouvrement;
use App\Models\Epargne;
use Illuminate\Support\Str;

class EpargneService
{
    /**
     * Enregistrer l'épargne quotidienne lors d'un encaissement.
     * L'épargne va UNIQUEMENT dans la caisse, pas en comptabilité.
     */
    public static function collecter(Vente $vente, Recouvrement $recouvrement): void
    {
        if ($vente->epargne_par_jour <= 0) return;

        Epargne::create([
            'uuid' => Str::uuid(),
            'vente_id' => $vente->id,
            'debiteur_id' => $vente->debiteur_id,
            'montant' => $vente->epargne_par_jour,
            'date_collecte' => $recouvrement->date_recouvrement,
            'statut' => 'collecte',
        ]);

        // Enregistrer dans la caisse (pas en comptabilité)
        CaisseService::enregistrerEpargne($vente, $vente->epargne_par_jour);
    }

    /**
     * La gérante récupère l'épargne d'un client.
     */
    public static function recuperer(Epargne $epargne, $userId): void
    {
        $epargne->update([
            'statut' => 'recuperee',
            'recuperee_par' => $userId,
            'date_recuperation' => now(),
        ]);
    }

    /**
     * Le client confirme avoir récupéré sa mise.
     */
    public static function confirmer(Epargne $epargne): void
    {
        $epargne->update(['confirmation_client' => true]);
    }

    /**
     * Total épargne collectée (non récupérée) pour la caisse.
     */
    public static function totalCollecte(): float
    {
        return Epargne::where('statut', 'collecte')->sum('montant');
    }
}
--- app/Services/PenaliteService.php ---
<?php
namespace App\Services;
use App\Models\Vente;
use App\Models\Recouvrement;
use App\Models\Echeance;
use App\Models\Penalite;
use Illuminate\Support\Str;

class PenaliteService
{
    /**
     * Vérifier et appliquer les pénalités pour les échéances impayées.
     * Les pénalités vont en caisse ET en comptabilité (divers).
     */
    public static function appliquer(Vente $vente, Recouvrement $recouvrement): void
    {
        $aujourdhui = now()->toDateString();
        
        $echeancesRetard = Echeance::where('vente_id', $vente->id)
            ->whereDate('date_echeance', '<', $aujourdhui)
            ->whereNotIn('statut', ['paye'])
            ->get();

        foreach ($echeancesRetard as $echeance) {
            $jours = $echeance->date_echeance->diffInDays(now());
            if ($jours <= 0) continue;

            $montant = $jours * $vente->penalite_par_jour;

            // Ne pas créer de doublon pour la même échéance
            $existe = Penalite::where('echeance_id', $echeance->id)->exists();
            if ($existe) continue;

            $penalite = Penalite::create([
                'uuid' => Str::uuid(),
                'vente_id' => $vente->id,
                'echeance_id' => $echeance->id,
                'recouvrement_id' => $recouvrement->id,
                'debiteur_id' => $vente->debiteur_id,
                'montant' => $montant,
                'jours_retard' => $jours,
                'date_appliquee' => $aujourdhui,
            ]);

            // Enregistrer en caisse (divers)
            CaisseService::enregistrerPenalite($penalite);
            
            // Enregistrer en comptabilité (divers)
            ComptabiliteService::ecrirePenalite($penalite);
        }
    }
}

<?php
namespace App\Models;
use Illuminate\Database\Eloquent\Model;

class Epargne extends Model
{
    protected $fillable = [
        'uuid', 'vente_id', 'debiteur_id', 'montant',
        'date_collecte', 'statut', 'recuperee_par',
        'date_recuperation', 'confirmation_client',
    ];

    protected $casts = [
        'date_collecte' => 'date',
        'date_recuperation' => 'date',
        'confirmation_client' => 'boolean',
    ];

    public function vente() { return $this->belongsTo(Vente::class); }
    public function debiteur() { return $this->belongsTo(Debiteur::class); }
    public function recupereePar() { return $this->belongsTo(User::class, 'recuperee_par'); }
}
