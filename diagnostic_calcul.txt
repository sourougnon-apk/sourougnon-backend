<?php

namespace App\Services;

use App\Models\Recouvrement;
use App\Models\Echeance;
use App\Models\Vente;

class PaiementService
{
    /**
     * Traiter un paiement selon le statut choisi par l'agent.
     * Gère les excédents : un paiement peut couvrir plusieurs échéances.
     */
    public static function affecterPaiement(Recouvrement $recouvrement): void
    {
        $vente = $recouvrement->vente;
        $aujourdhui = now()->toDateString();

        // Statuts non payants : absent, refus, promesse
        if (!in_array($recouvrement->statut, ['paye', 'partiel'])) {
            $echeance = Echeance::where('vente_id', $vente->id)
                ->whereDate('date_echeance', $aujourdhui)
                ->first()
                ?? Echeance::where('vente_id', $vente->id)
                    ->whereIn('statut', ['en_attente', 'en_retard', 'partiel'])
                    ->orderBy('date_echeance')
                    ->first();

            if ($echeance) {
                $echeance->statut = match ($recouvrement->statut) {
                    'absent' => 'absent',
                    'refus' => 'refus',
                    'promesse' => 'promesse',
                    default => $echeance->statut,
                };
                $echeance->save();
            }

            $vente->mettreAJourStatutGlobal();
            return;
        }

        // Paiements réels : paye ou partiel
        $montantRestant = $recouvrement->montant;

        // 1. Échéances échues non soldées (retards)
        $echeancesDues = Echeance::where('vente_id', $vente->id)
            ->whereIn('statut', ['en_retard', 'partiel', 'en_attente'])
            ->whereDate('date_echeance', '<=', $aujourdhui)
            ->orderBy('date_echeance')
            ->get();

        foreach ($echeancesDues as $echeance) {
            if ($montantRestant <= 0) break;
            $reste = $echeance->restant();
            if ($reste <= 0) continue;

            $payer = min($montantRestant, $reste);
            $echeance->montant_paye += $payer;
            $echeance->date_paiement = $recouvrement->date_recouvrement;
            $echeance->jours_retard = 0;
            $echeance->statut = $echeance->montant_paye >= $echeance->montant_prevu ? 'paye' : 'partiel';
            $echeance->save();

            $montantRestant -= $payer;
        }

        // 2. Excédent vers les échéances futures
        if ($montantRestant > 0) {
            $echeancesFutures = Echeance::where('vente_id', $vente->id)
                ->where('statut', 'en_attente')
                ->whereDate('date_echeance', '>', $aujourdhui)
                ->orderBy('date_echeance')
                ->get();

            foreach ($echeancesFutures as $echeance) {
                if ($montantRestant <= 0) break;
                $reste = $echeance->restant();
                if ($reste <= 0) continue;

                $payer = min($montantRestant, $reste);
                $echeance->montant_paye += $payer;
                $echeance->date_paiement = $recouvrement->date_recouvrement;
                $echeance->statut = $echeance->montant_paye >= $echeance->montant_prevu ? 'paye' : 'partiel';
                $echeance->save();

                $montantRestant -= $payer;
            }
        }

        // Mettre à jour le statut global de la vente
        $vente->mettreAJourStatutGlobal();

        // Épargne et pénalités uniquement en cas de paiement complet
        if ($recouvrement->statut === 'paye') {
            EpargneService::collecter($vente, $recouvrement);
            PenaliteService::appliquer($vente, $recouvrement);
        }
    }
}

<?php
namespace App\Services;
use App\Models\Vente;
use App\Models\Recouvrement;
use App\Models\Epargne;
use Illuminate\Support\Str;

class EpargneService
{
    /**
     * Enregistrer l'épargne quotidienne lors d'un encaissement.
     * L'épargne va UNIQUEMENT dans la caisse, pas en comptabilité.
     */
    public static function collecter(Vente $vente, Recouvrement $recouvrement): void
    {
        if ($vente->epargne_par_jour <= 0) return;

        Epargne::create([
            'uuid' => Str::uuid(),
            'vente_id' => $vente->id,
            'debiteur_id' => $vente->debiteur_id,
            'montant' => $vente->epargne_par_jour,
            'date_collecte' => $recouvrement->date_recouvrement,
            'statut' => 'collecte',
        ]);

        // Enregistrer dans la caisse (pas en comptabilité)
        CaisseService::enregistrerEpargne($vente, $vente->epargne_par_jour);
    }

    /**
     * La gérante récupère l'épargne d'un client.
     */
    public static function recuperer(Epargne $epargne, $userId): void
    {
        $epargne->update([
            'statut' => 'recuperee',
            'recuperee_par' => $userId,
            'date_recuperation' => now(),
        ]);
    }

    /**
     * Le client confirme avoir récupéré sa mise.
     */
    public static function confirmer(Epargne $epargne): void
    {
        $epargne->update(['confirmation_client' => true]);
    }

    /**
     * Total épargne collectée (non récupérée) pour la caisse.
     */
    public static function totalCollecte(): float
    {
        return Epargne::where('statut', 'collecte')->sum('montant');
    }
}

<?php
namespace App\Services;
use App\Models\Notification;
use App\Models\Echeance;
use App\Models\Vente;

use Illuminate\Support\Str;

class NotificationService
{
    public static function generer(): array
    {
        $nb = 0;
        $aujourdhui = now()->toDateString();

        // 1. Échéances du jour — notifier les agents
        $echeancesJour = Echeance::whereDate('date_echeance', $aujourdhui)
            ->where('statut', 'en_attente')
            ->with('vente.agent')
            ->get();
        foreach ($echeancesJour as $e) {
            $agentId = $e->vente->agent_id;
            $existe = Notification::where('user_id', $agentId)
                ->where('type', 'echeance')
                ->whereDate('created_at', $aujourdhui)
                ->where('message', 'LIKE', '%' . $e->vente->debiteur?->nom . '%')
                ->exists();
            if (!$existe) {
                Notification::create([
                    'uuid' => Str::uuid(),
                    'user_id' => $agentId,
                    'type' => 'echeance',
                    'titre' => 'Échéance du jour',
                    'message' => ($e->vente->debiteur?->nom ?? 'Débiteur') . ' doit payer ' . number_format($e->montant_prevu, 0, ',', ' ') . ' FCFA aujourd\'hui.',
                    'lien' => '#agent-space',
                ]);
                $nb++;
            }
        }

        // 2. Retards — notifier les agents
        $retards = Echeance::whereIn('statut', ['en_retard', 'partiel'])
            ->whereDate('date_echeance', '<', $aujourdhui)
            ->with('vente.agent')
            ->get();
        foreach ($retards as $r) {
            $agentId = $r->vente->agent_id;
            $existe = Notification::where('user_id', $agentId)
                ->where('type', 'retard')
                ->whereDate('created_at', $aujourdhui)
                ->where('message', 'LIKE', '%' . $r->vente->debiteur?->nom . '%')
                ->exists();
            if (!$existe) {
                Notification::create([
                    'uuid' => Str::uuid(),
                    'user_id' => $agentId,
                    'type' => 'retard',
                    'titre' => 'Retard de paiement',
                    'message' => ($r->vente->debiteur?->nom ?? 'Débiteur') . ' a ' . $r->jours_retard . ' jour(s) de retard. Restant : ' . number_format($r->restant(), 0, ',', ' ') . ' FCFA.',
                    'lien' => '#agent-space',
                ]);
                $nb++;
            }
        }

        // 3. Stocks faibles — notifier la gérante
        $stocksFaibles = Produit::where('actif', true)->where('stock', '<=', \DB::raw('seuil_alerte'))->where(function($q){ $q->where('stock', '>', 0)->orWhere('stock_initial', '>', 0); })->get();
        $gerante = \App\Models\User::where('role', 'gerante')->first();
        if ($gerante && $stocksFaibles->isNotEmpty()) {
            foreach ($stocksFaibles as $p) {
                $existe = Notification::where('user_id', $gerante->id)
                    ->where('type', 'stock')
                    ->whereDate('created_at', $aujourdhui)
                    ->where('message', 'LIKE', '%' . $p->nom . '%')
                    ->exists();
                if (!$existe) {
                    Notification::create([
                        'uuid' => Str::uuid(),
                        'user_id' => $gerante->id,
                        'type' => 'stock',
                        'titre' => 'Stock faible',
                        'message' => $p->nom . ' : ' . $p->stock . ' unité(s) (seuil : ' . $p->seuil_alerte . ')',
                        'lien' => '#stocks',
                    ]);
                    $nb++;
                }
            }
        }

        return ['nb_notifications' => $nb];
    }

    // Lien WhatsApp pour une échéance
    public static function lienWhatsapp(Echeance $echeance): string
    {
        $tel = $echeance->vente->debiteur?->telephone;
        if (!$tel) return '';
        $nom = $echeance->vente->debiteur?->nom ?? 'Client';
        $msg = "Bonjour $nom,\nNous vous rappelons votre échéance de " 
            . number_format($echeance->montant_prevu, 0, ',', ' ') 
            . " FCFA.\nRestant dû : " 
            . number_format($echeance->restant(), 0, ',', ' ') 
            . " FCFA.\nMerci de procéder au règlement.\nSourougnon";
        return 'https://wa.me/' . preg_replace('/[^0-9]/', '', $tel) . '?text=' . urlencode($msg);
    }

    public function encaisser(Request $request)
    {
        $request->validate([
            'vente_id' => 'required|exists:ventes,uuid',
            'montant' => 'required|numeric|min:0',
            'mode_paiement' => 'required|in:especes,mobile,autre',
            'statut' => 'required|in:paye,partiel,absent,refus,promesse',
            'commentaire' => 'nullable|string|required_if:statut,absent,refus,promesse',
        ]);
        
        $vente = Vente::where('uuid', $request->vente_id)->firstOrFail();
        
        $rec = Recouvrement::create([
            'uuid' => Str::uuid(),
            'vente_id' => $vente->id,
            'agent_id' => $vente->debiteur->agent_id ?? $request->user()->id,
            'montant' => $request->montant,
            'date_recouvrement' => now()->toDateString(), 'heure_recouvrement' => now()->toTimeString(),
            'mode_paiement' => $request->mode_paiement,
            'statut' => $request->statut,
            'commentaire' => $request->commentaire,
            'synced' => true,
        ]);

        // Traiter le paiement avec le service
        PaiementService::affecterPaiement($rec);

        // Encaissements réels : caisse + compta
        if (in_array($request->statut, ['paye', 'partiel']) && $request->montant > 0) {
            CaisseService::enregistrerRecouvrement($rec);
            ComptabiliteService::ecrireRecouvrement($rec);
        }

        // Notifications
        NotificationService::generer();

        return response()->json([
            'success' => true,
            'reste' => $vente->resteAPayer(),
            'message' => self::messageStatut($request->statut),
        ]);
    }
